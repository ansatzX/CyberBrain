# Goal v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 pi 的 goal 系统从 v1（模型直接写文件 + 每轮注入）重构为 v2（状态机 owner + 工具入口 + 事件驱动注入），对齐 codex `ext/goal/` 架构。

**Architecture:** 纯逻辑层（`goal-core.ts`，可单测）与 pi 扩展壳（`goal.ts`）分离。存储 = per-session 文件（`~/.pi/agent/goals/<会话哈希>.md`）。模型只能通过三个工具（get_goal/create_goal/update_goal）交互，扩展代码是唯一写文件者。注入 = 事件驱动（objective_updated/continuation），custom message + context 过滤，日常零注入。

**Tech Stack:** TypeScript（pi 扩展，`@earendil-works/pi-coding-agent` API）、node:test（单测，node 24 type stripping）、node:fs、node:crypto。

## Global Constraints

- **pi 扩展语法约束**：解析器不支持 `??` 和 `?.`——一律用 `||`/`&&`/三元改写（`!a?.b` 写成 `!a || !b`）
- **不引入外部依赖**：只用 node 内置模块 + pi 类型 import
- 状态字段：`status` 取值仅 `active | blocked | complete | abandoned`
- 无预算字段（设计已砍）
- 模型永不直接写 goals 文件
- `~/.pi/agent` 非 git 仓库：任务完成标志 = 测试/验证通过（无 commit 步骤）

```typescript
// goal-core.ts — goal 纯逻辑层（无 pi API 依赖，可单测）
// 存储：per-session 文件 ~/.pi/agent/goals/<sha1(threadId)>.md
// 注意 pi 扩展解析器不支持 ?? / ?. —— 全部用 || / && / 三元

import { createHash } from "node:crypto";
import { mkdirSync, readFileSync, writeFileSync, existsSync } from "node:fs";
import { homedir } from "node:os";
import { join } from "node:path";

export interface GoalState {
	thread_id: string;
	goal_id: string;
	objective: string;
	status: "active" | "blocked" | "complete" | "abandoned";
	created_at: string;
	updated_at: string;
	blocked_streak: number;
	blocked_condition: string | null;
	complete_reason: string | null;
}

export type GoalStatus = GoalState["status"];

const VALID_STATUS: string[] = ["active", "blocked", "complete", "abandoned"];

function goalsDir(): string {
	const override = process.env.PI_GOAL_TEST_DIR;
	return override || join(homedir(), ".pi", "agent", "goals");
}

export function goalFilePath(threadId: string): string {
	const hash = createHash("sha1").update(threadId).digest("hex").slice(0, 16);
	return join(goalsDir(), `${hash}.md`);
}

function parseGoal(raw: string): GoalState | null {
	const fm = raw.match(/^---\n([\s\S]*?)\n---\n/);
	if (!fm) return null;
	const get = (key: string): string => {
		const m = fm[1].match(new RegExp(`^${key}:\\s*(.*)$`, "m"));
		return m ? m[1].trim() : "";
	};
	const status = get("status");
	if (!VALID_STATUS.includes(status)) return null;
	const stripQuotes = (s: string): string => s.replace(/^"(.*)"$/, "$1");
	return {
		thread_id: stripQuotes(get("thread_id")),
		goal_id: get("goal_id"),
		objective: stripQuotes(get("objective")),
		status: status as GoalStatus,
		created_at: get("created_at"),
		updated_at: get("updated_at"),
		blocked_streak: Number(get("blocked_streak")) || 0,
		blocked_condition: get("blocked_condition") ? stripQuotes(get("blocked_condition")) : null,
		complete_reason: get("complete_reason") ? stripQuotes(get("complete_reason")) : null,
	};
}

function serializeGoal(g: GoalState): string {
	const lines = [
		"---",
		`thread_id: ${g.thread_id}`,
		`goal_id: ${g.goal_id}`,
		`status: ${g.status}`,
		`created_at: ${g.created_at}`,
		`updated_at: ${g.updated_at}`,
		`blocked_streak: ${g.blocked_streak}`,
		`blocked_condition: ${g.blocked_condition || ""}`,
		`complete_reason: ${g.complete_reason || ""}`,
		`objective: ${g.objective.replace(/\n/g, " ")}`,
		"---",
		"",
	];
	return lines.join("\n");
}

export function loadGoal(threadId: string): GoalState | null {
	const file = goalFilePath(threadId);
	if (!existsSync(file)) return null;
	const raw = readFileSync(file, "utf8");
	return parseGoal(raw);
}

export function saveGoal(g: GoalState): void {
	mkdirSync(goalsDir(), { recursive: true });
	writeFileSync(goalFilePath(g.thread_id), serializeGoal(g), "utf8");
}

function newId(): string {
	return `${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 10)}`;
}

export function createGoal(threadId: string, objective: string): GoalState {
	const existing = loadGoal(threadId);
	if (existing && existing.status === "active") {
		throw new Error("An active goal already exists. Clear it first (/ansatz:goal clear).");
	}
	const now = new Date().toISOString();
	const g: GoalState = {
		thread_id: threadId,
		goal_id: newId(),
		objective,
		status: "active",
		created_at: now,
		updated_at: now,
		blocked_streak: 0,
		blocked_condition: null,
		complete_reason: null,
	};
	saveGoal(g);
	return g;
}

export function archiveGoal(threadId: string, reason: string): GoalState | null {
	const g = loadGoal(threadId);
	if (!g) return null;
	g.status = "abandoned";
	g.updated_at = new Date().toISOString();
	g.complete_reason = reason || "cleared via /ansatz:goal clear";
	saveGoal(g);
	return g;
}

export function isRecentUpdate(g: GoalState, minAgeMs: number): boolean {
	try {
		return Date.now() - Date.parse(g.updated_at) < minAgeMs;
	} catch {
		return false;
	}
}

export function touchGoal(g: GoalState): void {
	g.updated_at = new Date().toISOString();
	saveGoal(g);
}
```

```typescript
export type UpdateGoalOutcome = { ok: true; goal: GoalState } | { ok: false; error: string };

export function nextBlockedStreak(current: number, prevCondition: string | null, condition: string): number {
	if (prevCondition !== null && prevCondition === condition) return current + 1;
	return 1;
}

export function registerBlockedOccurrence(threadId: string, condition: string): void {
	const g = loadGoal(threadId);
	if (!g || g.status !== "active") return;
	g.blocked_streak = nextBlockedStreak(g.blocked_streak, g.blocked_condition, condition);
	g.blocked_condition = condition;
	g.updated_at = new Date().toISOString();
	saveGoal(g);
}

export function updateGoalStatus(
	threadId: string,
	status: "complete" | "blocked",
	reason: string,
): UpdateGoalOutcome {
	const g = loadGoal(threadId);
	if (!g) return { ok: false, error: "No goal exists for this session." };
	if (g.status !== "active") return { ok: false, error: `Goal is already ${g.status}.` };
	if (!reason.trim()) return { ok: false, error: "A reason is required when updating goal status." };

	if (status === "complete") {
		g.status = "complete";
		g.complete_reason = reason.trim();
		g.updated_at = new Date().toISOString();
		saveGoal(g);
		return { ok: true, goal: g };
	}

	// blocked
	if (g.blocked_streak < 3) {
		return {
			ok: false,
			error: `Same blocking condition must recur for at least 3 consecutive goal turns before marking blocked (currently ${g.blocked_streak}/3).`,
		};
	}
	g.status = "blocked";
	g.complete_reason = reason.trim();
	g.updated_at = new Date().toISOString();
	saveGoal(g);
	return { ok: true, goal: g };
}
```

```typescript
export function objectiveUpdatedPrompt(g: GoalState): string {
	return [
		`新持久化目标已设置：${g.objective}`,
		"",
		"行为要求：",
		"- 从当前工作区证据出发推进目标，不要重定义成功标准。",
		"- 目标达成后调用 update_goal 工具，status=\"complete\"，附理由。",

	return [
		`继续推进持久化目标：${g.objective}`,
		"",
		"行为要求：",
		"- 目标跨轮次持续存在；本轮回合不必缩小目标范围。",
		"- 以当前工作区和外部状态为权威证据，先检查现状再决定下一步。",
```

```typescript
// goal.ts — goal v2 扩展壳（命令层 + 工具层 + 事件层）
// 架构：goal-core.ts 纯逻辑 + 本文件 pi API 挂接
// 注意：解析器不支持 ?? / ?.

import type { ExtensionAPI } from "@earendil-works/pi-coding-agent";
import { createHash } from "node:crypto";
import {
	loadGoal, createGoal, archiveGoal, touchGoal, updateGoalStatus,
	objectiveUpdatedPrompt, continuationPrompt, type GoalState,
} from "./goal-core.ts";

export function getSessionThreadId(sessionFile: string | undefined): string {
	if (!sessionFile) return "ephemeral";
	return createHash("sha1").update(sessionFile).digest("hex").slice(0, 16);
}

/** 待注入的一次性 steering（objective_updated），before_agent_start 消费后清空 */
let pendingSteering: { threadId: string; prompt: string } | null = null;

export default function goalExtension(pi: ExtensionAPI) {
	const threadIdFor = (ctx: { sessionManager: { getSessionFile(): string | undefined } }): string =>
		getSessionThreadId(ctx.sessionManager.getSessionFile());

	// ---------- 命令层 ----------
	pi.registerCommand("ansatz:goal", {
		description: "Set or view the goal for a long-running task (like Codex /goal)",
		getArgumentCompletions: (prefix: string) => {
			const subs = ["set", "view", "clear"].filter((s) => s.startsWith(prefix.toLowerCase()));
			return subs.length > 0 ? subs.map((s) => ({ value: s, label: s })) : null;
		},
		handler: async (args, ctx) => {
			const trimmed = args.trim();
			const lower = trimmed.toLowerCase();
			const threadId = threadIdFor(ctx);

			// view
			if (!trimmed || lower === "view") {
				const g = loadGoal(threadId);
				if (!g || g.status !== "active") {
					ctx.ui.notify("No active goal. Use /ansatz:goal set <objective> to create one.", "info");
					return;
				}
				const items = [
					`Goal: ${g.objective}`,
					`Status: ${g.status}`,
					`Created: ${g.created_at}`,
					`Updated: ${g.updated_at}`,
					`Blocked streak: ${g.blocked_streak}`,
				];
				if (ctx.hasUI) {
					await ctx.ui.select("Current Goal", items);
				} else {
					ctx.ui.notify(items.join("\n"), "info");
				}
				return;
			}

			// clear
			if (lower === "clear" || lower.startsWith("clear ")) {
				const reason = trimmed.slice("clear".length).trim();
				const g = archiveGoal(threadId, reason);
				if (!g) {
					ctx.ui.notify("No goal to clear.", "info");
					return;
				}
				ctx.ui.notify("Goal cleared (abandoned).", "info");
				return;
			}

			// set
			const text = lower.startsWith("set") ? trimmed.slice(3).trim() : trimmed;
			if (!text) {
				ctx.ui.notify("Usage: /ansatz:goal set <objective>", "error");
				return;
			}
			let goal: GoalState;
			try {
				goal = createGoal(threadId, text);
			} catch (e) {
				ctx.ui.notify((e as Error).message, "error");
				return;
			}
			ctx.ui.notify(`Goal set: ${text}`, "info");
			pendingSteering = { threadId, prompt: objectiveUpdatedPrompt(goal) };
			pi.sendUserMessage(objectiveUpdatedPrompt(goal));
		},
	});
}
```

```typescript
	pi.registerTool({
		name: "get_goal",
		label: "Get goal",
		description: "Get the current goal for this session, including status and blocked audit state.",
		parameters: {},
		async execute(_toolCallId, _params, _signal, _onUpdate, ctx) {
			const threadId = threadIdFor(ctx);
			const g = loadGoal(threadId);
			if (!g) {
				return { content: [{ type: "text", text: "No goal set for this session." }], details: {} };
			}
			return {

						`updated_at: ${g.updated_at}`,

						`blocked_condition: ${g.blocked_condition || "none"}`,
					].join("\n"),
				}],
				details: {},
			};
		},
	});

	pi.registerTool({
		name: "create_goal",
		label: "Create goal",
		description: "Create a goal only when explicitly requested by the user or system/developer instructions; do not infer goals from ordinary tasks. Fails if an unfinished goal exists.",
		parameters: {
			type: "object",
			properties: { objective: { type: "string", description: "Required. The concrete objective to start pursuing." } },
			required: ["objective"],
		},
		async execute(_toolCallId, params, _signal, _onUpdate, ctx) {
			const threadId = threadIdFor(ctx);
			const objective = (params.objective as string) || "";
			if (!objective.trim()) {
				return { content: [{ type: "text", text: "error: objective is required" }], details: {} };
			}
			try {
				const g = createGoal(threadId, objective.trim());
				pendingSteering = { threadId, prompt: objectiveUpdatedPrompt(g) };
				return { content: [{ type: "text", text: `Goal created: ${g.objective}` }], details: {} };
			} catch (e) {
				return { content: [{ type: "text", text: `error: ${(e as Error).message}` }], details: {} };
			}
		},
	});

	pi.registerTool({
		name: "update_goal",
		label: "Update goal",
		description: "Update the existing goal. Use only to mark the goal achieved (status=complete, when the objective is actually achieved and no required work remains) or genuinely blocked (status=blocked, only after the same blocking condition has recurred for at least three consecutive goal turns). Do not use for hard/slow/uncertain work. Always provide a reason.",
		parameters: {
			type: "object",
			properties: {
				status: { type: "string", enum: ["complete", "blocked"], description: "complete or blocked" },
				reason: { type: "string", description: "Required. Why this status applies." },
			},
			required: ["status", "reason"],
		},
		async execute(_toolCallId, params, _signal, _onUpdate, ctx) {
			const threadId = threadIdFor(ctx);
			const status = params.status as "complete" | "blocked";
			const reason = (params.reason as string) || "";
			if (status !== "complete" && status !== "blocked") {
				return { content: [{ type: "text", text: "error: status must be complete or blocked" }], details: {} };
			}
			const result = updateGoalStatus(threadId, status, reason);
			if (!result.ok) {
				return { content: [{ type: "text", text: `error: ${result.error}` }], details: {} };
			}
			ctx.ui.notify(`Goal marked ${result.goal.status}: ${reason}`, "info");
			return {
				content: [{ type: "text", text: `Goal ${status}: ${result.goal.complete_reason}` }],
				details: {},
			};
		},
	});

	// ---------- 注入层（事件驱动） ----------
	// 每轮注入：仅当有 pending steering（objective_updated），消费后清空
	pi.on("before_agent_start", async () => {
		if (!pendingSteering) return undefined;
		const { prompt } = pendingSteering;
		pendingSteering = null;
		return {
			message: { customType: "goal-context", content: prompt, display: false },
		};
	});

	// context 过滤：历史里只保留最新一条 goal-context（防膨胀）
	pi.on("context", async (event) => {
		const msgs = event.messages as Array<{ customType?: string; timestamp?: string }>;
		let lastGoalIdx = -1;
		for (let i = 0; i < msgs.length; i++) {
			if (msgs[i].customType === "goal-context") lastGoalIdx = i;
		}
		if (lastGoalIdx === -1) return undefined;
		return {
			messages: event.messages.filter((_m, i) => {
				const m = msgs[i];
				return m.customType !== "goal-context" || i === lastGoalIdx;
			}),
		};
	});

	// ---------- 续跑层（对齐 codex on_thread_idle） ----------
	pi.on("agent_settled", async (_event, ctx) => {
		const threadId = threadIdFor(ctx);
		const g = loadGoal(threadId);
		if (!g || g.status !== "active") return;
		if (isRecentUpdate(g, 5 * 60 * 1000)) return; // 5 分钟冷却
		touchGoal(g); // 续跑也视为一次"更新"，避免无限循环
		ctx.ui.notify("Goal auto-continue: nudging agent", "info");
		pi.sendUserMessage(continuationPrompt(g), { deliverAs: "followUp" });
	});
```
